const SHEET_NAME = "Sheet1"; // ===================================================== // GET // ===================================================== function doGet(e) { const action = e.parameter.action || "getData"; const callback = e.parameter.callback || "callback"; try { const sheet = getSheet(); // ================================================= // GET BUTTON TEXT REALTIME (ดึงข้อมูลข้อความเมื่อกดปุ่ม) // ================================================= if (action === "getButtonText") { const type = e.parameter.type || ""; const index = Number(e.parameter.index) || 0; const cell = e.parameter.cell || ""; let text = ""; if (cell) { text = sheet.getRange(cell).getValue() || ""; } else if (type === "quick") { text = sheet.getRange(index + 2, 2).getValue() || ""; // B2:B17 } else if (type === "teamQuick1") { text = sheet.getRange(index + 19, 2).getValue() || ""; // B19:B24 } else if (type === "teamQuick2") { text = sheet.getRange(index + 14, 7).getValue() || ""; // G14:G19 } else if (type === "ytDesc") { text = sheet.getRange(index + 12, 2).getValue() || ""; // B12:B15 } else if (type === "crazy") { text = sheet.getRange(index + 7, 7).getValue() || ""; // G7:G9 } else if (type === "crazyCover") { text = sheet.getRange(index + 12, 7).getValue() || ""; // G12:G13 } else if (type === "germany") { text = sheet.getRange(index + 10, 2).getValue() || ""; // B10:B11 } return jsonp({ success: true, text: text }, callback); } // ================================================= // GET DATA (ดึงเฉพาะชื่อปุ่มตอนโหลดหน้าแรก) // ================================================= if (action === "getData") { // ดึง A2:B39 const valuesA_B = sheet.getRange("A2:B39").getValues(); // ดึง F7:G19 const valuesF_G = sheet.getRange("F7:G19").getValues(); // ----------------------------------------------- // A2:A17 - เมนูด่วน 16 ปุ่ม (เก็บเฉพาะชื่อ) // ----------------------------------------------- const buttons = valuesA_B.slice(0, 16).map(function(row, idx) { return { name: row[0] || "", type: "quick", index: idx }; }); // ----------------------------------------------- // A6:B6 - ปุ่มด่วนหน้า รวมชื่อทีม // ----------------------------------------------- const btnA6_B6 = { name: valuesA_B[4][0] || "", cell: "B6" }; // ----------------------------------------------- // A7:B7 - ปุ่มด่วน Shorts AI 1 // ----------------------------------------------- const btnA7_B7 = { name: valuesA_B[5][0] || "", cell: "B7" }; // ----------------------------------------------- // A8:B8 - ปุ่มด่วน บอล AI // ----------------------------------------------- const btnA8_B8 = { name: valuesA_B[6][0] || "", cell: "B8" }; // ----------------------------------------------- // A39:B39 - ปุ่มด่วน Shorts AI 3 (ใหม่) // ----------------------------------------------- const btnA39_B39 = { name: valuesA_B[37][0] || "", cell: "B39" }; // ----------------------------------------------- // A10:B11 - ปุ่มด่วน Prompt ปก Germany 2 ปุ่ม // ----------------------------------------------- const germanyButtons = valuesA_B.slice(8, 10).map(function(row, idx) { return { name: row[0] || "", type: "germany", index: idx }; }); // ----------------------------------------------- // A12:B15 - ปุ่มด่วน YouTube Description 4 ปุ่ม // ----------------------------------------------- const ytDescButtons = valuesA_B.slice(10, 14).map(function(row, idx) { return { name: row[0] || "", type: "ytDesc", index: idx }; }); // ----------------------------------------------- // F7:G9 - ปุ่มด่วน บอลบ้า 3 ปุ่ม (ดึงชื่อ F คัดลอก G) // ----------------------------------------------- const crazyButtons = [ { name: valuesF_G[0][0] || "", type: "crazy", index: 0 }, { name: valuesF_G[1][0] || "", type: "crazy", index: 1 }, { name: valuesF_G[2][0] || "", type: "crazy", index: 2 } ]; // ----------------------------------------------- // F12:G13 - ปุ่มด่วน Prompt ปกบอลบ้า 2 ปุ่ม // ----------------------------------------------- const crazyCoverButtons = [ { name: valuesF_G[5][0] || "", type: "crazyCover", index: 0 }, { name: valuesF_G[6][0] || "", type: "crazyCover", index: 1 } ]; // ----------------------------------------------- // F10:G10 - ปุ่มด่วน Shorts AI 2 // ----------------------------------------------- const btnF10_G10 = { name: valuesF_G[3][0] || "", cell: "G10" }; // ----------------------------------------------- // A19:A24 - ปุ่มด่วน รวมชื่อทีม แถวที่ 1 (6 ปุ่ม) // ----------------------------------------------- const teamQuickButtons1 = valuesA_B.slice(17, 23).map(function(row, idx) { return { name: row[0] || "", type: "teamQuick1", index: idx }; }); // ----------------------------------------------- // F14:F19 - ปุ่มด่วน รวมชื่อทีม แถวที่ 2 (6 ปุ่ม) // ----------------------------------------------- const teamQuickButtons2 = valuesF_G.slice(7, 13).map(function(row, idx) { return { name: row[0] || "", type: "teamQuick2", index: idx }; }); return jsonp({ success: true, buttons: buttons, btnA6_B6: btnA6_B6, btnA7_B7: btnA7_B7, btnA8_B8: btnA8_B8, btnA39_B39: btnA39_B39, germanyButtons: germanyButtons, ytDescButtons: ytDescButtons, crazyButtons: crazyButtons, crazyCoverButtons: crazyCoverButtons, btnF10_G10: btnF10_G10, teamQuickButtons1: teamQuickButtons1, teamQuickButtons2: teamQuickButtons2 }, callback); } // ================================================= // IMAGE // ================================================= if (action === "image") { const imageUrl = e.parameter.url; if (!imageUrl) { return jsonp({ success: false, error: "ไม่พบ URL รูปภาพ" }, callback); } const response = UrlFetchApp.fetch(imageUrl, { muteHttpExceptions: true, followRedirects: true }); const code = response.getResponseCode(); if (code !== 200) { return jsonp({ success: false, error: "ไม่สามารถโหลดรูปภาพได้ HTTP " + code }, callback); } const blob = response.getBlob(); const base64 = Utilities.base64Encode(blob.getBytes()); const mime = blob.getContentType() || "image/png"; return jsonp({ success: true, mime: mime, base64: base64 }, callback); } // ================================================= // UNKNOWN ACTION // ================================================= return jsonp({ success: false, error: "ไม่รู้จัก action" }, callback); } catch (error) { return jsonp({ success: false, error: error.toString() }, callback); } } // ===================================================== // POST // ===================================================== function doPost(e) { try { const sheet = getSheet(); const targetCell = e.parameter.cell || ""; const value = e.parameter.value || ""; if (targetCell) { sheet.getRange(targetCell).setValue(value); SpreadsheetApp.flush(); return ContentService .createTextOutput(JSON.stringify({ success: true, cell: targetCell, value: value })) .setMimeType(ContentService.MimeType.JSON); } const row = Number(e.parameter.row); if (!row || row < 18 || row > 39) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: "Row ต้องอยู่ระหว่าง 18-39 หรือระบุ cell" })) .setMimeType(ContentService.MimeType.JSON); } sheet.getRange(row, 2).setValue(value); SpreadsheetApp.flush(); return ContentService .createTextOutput(JSON.stringify({ success: true, row: row, value: value })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ===================================================== // GET SHEET // ===================================================== function getSheet() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); const sheet = spreadsheet.getSheetByName(SHEET_NAME); if (!sheet) { throw new Error("ไม่พบ Sheet ชื่อ: " + SHEET_NAME); } return sheet; } // ===================================================== // JSONP // ===================================================== function jsonp(data, callback) { const json = JSON.stringify(data); const safeCallback = callback.replace(/[^\w.$]/g, ""); return ContentService .createTextOutput(safeCallback + "(" + json + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); }
